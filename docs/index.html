صحيح جستجوی_دودویی(صحيح مقادیر[], صحيح شمار, صحيح کلید) {
  صحيح کم = 0, بالا = شمار - 1;
  طالما (کم <= بالا) {
    صحيح میانه = کم + (بالا - کم) / 2;
    إذا (مقادیر[میانه] == کلید) أرجع میانه;
    إذا (مقادیر[میانه] < کلید) کم = میانه + 1; وإلا بالا = میانه - 1;
  }
  أرجع -1;
}
